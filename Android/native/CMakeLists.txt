cmake_minimum_required(VERSION 3.18)
project(subrenamer_smb C)
if(NOT SMB2_SOURCE)
  message(FATAL_ERROR "Pass -DSMB2_SOURCE=<pinned libsmb2 source>")
endif()
set(ENABLE_LIBKRB5 OFF CACHE BOOL "" FORCE)
set(ENABLE_GSSAPI OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBDCERPC OFF CACHE BOOL "" FORCE)
set(ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ENABLE_UTILS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory("${SMB2_SOURCE}" smb2 EXCLUDE_FROM_ALL)
# Android packages unversioned .so names. Keep libsmb2 dynamically replaceable.
set_property(TARGET smb2 PROPERTY VERSION)
set_property(TARGET smb2 PROPERTY SOVERSION)
add_library(subrenamersmb SHARED subrenamer_smb.c)
target_include_directories(subrenamersmb PRIVATE "${SMB2_SOURCE}/include")
target_link_libraries(subrenamersmb PRIVATE smb2)
target_compile_options(subrenamersmb PRIVATE -Wall -Wextra -Werror)
